local function makeDraggable(title, frame)
	local dragging, dragStart, startPos = false, nil, nil
	title.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
		end
	end)
	title.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end
makeDraggable(panelTitle, panelFrame)
makeDraggable(eggPanelTitle, eggPanel)

local function getHum()
	local c = player.Character
	return c and c:FindFirstChildOfClass("Humanoid")
end

local function getRoot()
	local c = player.Character
	return c and (c:FindFirstChild("HumanoidRootPart") or c.PrimaryPart)
end

local function getJumpLaunchY(hum)
	if not hum then return 50 end
	if hum.UseJumpPower ~= false and tonumber(hum.JumpPower) and hum.JumpPower > 0 then
		return hum.JumpPower
	end
	local jh = tonumber(hum.JumpHeight)
	if jh and jh > 0 then
		return math.sqrt(2 * workspace.Gravity * jh)
	end
	return 50
end

local function launchHumanoid(hum, forceOverride)
	local root = getRoot()
	if not hum or not root then return end
	pcall(function()
		hum.Sit = false
		hum.PlatformStand = false
	end)
	local launchY = forceOverride or getJumpLaunchY(hum)
	local vel = root.AssemblyLinearVelocity
	root.AssemblyLinearVelocity = Vector3.new(vel.X, math.max(vel.Y, launchY), vel.Z)
end

local function instantStop()
	local root = getRoot()
	if root then
		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
	end
end

local function tpToGround()
	local root = getRoot()
	local hum = getHum()
	if not root or not hum then return end
	local origin = root.Position
	local rayParams = RaycastParams.new()
	rayParams.FilterDescendantsInstances = {character}
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	rayParams.IgnoreWater = true
	local result = workspace:Raycast(origin, Vector3.new(0, -10000, 0), rayParams)
	if result then
		root.CFrame = CFrame.new(result.Position + Vector3.new(0, 3, 0))
		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
	else
		root.CFrame = CFrame.new(origin.X, origin.Y - 50, origin.Z)
		root.AssemblyLinearVelocity = Vector3.zero
	end
end

local function rejoinServer()
	local success = pcall(function()
		TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, player)
	end)
	if not success then
		pcall(function()
			TeleportService:Teleport(game.PlaceId, player)
		end)
	end
end

function setSpeedLock(value)
	currentSpeedLock = value
	if SpeedConnection then
		SpeedConnection:Disconnect()
		SpeedConnection = nil
	end
	if value then
		SpeedConnection = RunService.Heartbeat:Connect(function()
			local hum = getHum()
			if hum and hum.Health > 0 then
				hum.WalkSpeed = value
			end
		end)
		setToggleVisual(speedLockToggle, true)
		speedBox.Text = tostring(value)
	else
		local hum = getHum()
		if hum then hum.WalkSpeed = 16 end
		setToggleVisual(speedLockToggle, false)
		speedBox.Text = "16"
	end
end

function setInfiniteJump(state)

local function toggleFPSBoost(enable)
	fpsBoostEnabled = enable
	setToggleVisual(fpsBoostToggle, enable)
	if enable then
		savedData = {}
		task.spawn(function()
			local BATCH = 250
			local n = 0
			local function yieldBatch()
				n = n + 1
				if n % BATCH == 0 then task.wait() end
			end
			local roots = {}
			local ms = game:FindFirstChild("MaterialService")
			if ms then table.insert(roots, ms) end
			for _, svcName in ipairs({"Workspace", "ReplicatedStorage", "Lighting"}) do
				local svc = game:FindFirstChild(svcName)
				if svc then table.insert(roots, svc) end
			end
			for _, root in ipairs(roots) do
				local ok, desc = pcall(function() return root:GetDescendants() end)
				if ok and desc then
					for _, obj in ipairs(desc) do
						if obj:IsA("MaterialVariant") or obj.ClassName == "MaterialVariant" then
							table.insert(savedData, {Object = obj, Kind = "MaterialVariant", Parent = obj.Parent})
							pcall(function() obj.Parent = nil end)
							yieldBatch()
						elseif obj:IsA("BasePart") then
							table.insert(savedData, {
								Object = obj, Kind = "Part",
								Material = obj.Material, CastShadow = obj.CastShadow, Reflectance = obj.Reflectance,
							})
							obj.Material = Enum.Material.SmoothPlastic
							obj.CastShadow = false
							obj.Reflectance = 0
							yieldBatch()
						elseif obj:IsA("Texture") or obj:IsA("Decal") then
							table.insert(savedData, {Object = obj, Kind = "Texture", Texture = obj.Texture})
							obj.Texture = ""
							yieldBatch()
						elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") then
							if obj.Name ~= "XeboBeam" then
								table.insert(savedData, {Object = obj, Kind = "Effect", Enabled = obj.Enabled})
								obj.Enabled = false
								yieldBatch()
							end
						end
					end
				end
			end
		end)
	else
		task.spawn(function()
			local BATCH = 300
			for i, data in ipairs(savedData) do
				local obj = data.Object
				if obj then
					if data.Kind == "MaterialVariant" then
						if data.Parent and data.Parent.Parent then
							pcall(function() obj.Parent = data.Parent end)
						end
					elseif obj.Parent then
						if data.Kind == "Part" then
							obj.Material = data.Material
							obj.CastShadow = data.CastShadow
							obj.Reflectance = data.Reflectance or 0
						elseif data.Kind == "Texture" then
							obj.Texture = data.Texture
						elseif data.Kind == "Effect" then
							obj.Enabled = data.Enabled
						end
					end
				end
				if i % BATCH == 0 then task.wait() end
			end
			savedData = {}
		end)
	end
end

-- ====================== SNIPE (speed 5000) ======================
local function doSnipe(eggModel)
	if snipeActive then
		snipeActive = false
		instantStop()
		return
	end
	if not eggModel or not eggModel.Parent then return end

	local eggPart = eggModel.PrimaryPart or eggModel:FindFirstChildWhichIsA("BasePart")
	if not eggPart then return end

	snipeActive = true
	local bigForce = SNIPE_SPEED
	local smallForce = 28

	task.spawn(function()
		while snipeActive do
			local root = getRoot()
			local hum = getHum()
			if not root or not hum or hum.Health <= 0 then break end
			if not eggModel.Parent then break end

			eggPart = eggModel.PrimaryPart or eggModel:FindFirstChildWhichIsA("BasePart")
			if not eggPart then break end

			if root.Position.Y >= eggPart.Position.Y - 1 then
				instantStop()
				break
			end

			launchHumanoid(hum, bigForce)
			task.wait(0.02)
		end

		if not snipeActive then
			instantStop()
			return
		end

		task.wait(0.08)
		instantStop()

		local startTime = tick()
		while snipeActive and (tick() - startTime) < 20 do
			local root = getRoot()
			local hum = getHum()
			if not root or not hum or hum.Health <= 0 then break end
			if not eggModel.Parent then break end

			eggPart = eggModel.PrimaryPart or eggModel:FindFirstChildWhichIsA("BasePart")
			if not eggPart then break end

			local myPos = root.Position
			local eggPos = eggPart.Position
			local dx = eggPos.X - myPos.X
			local dz = eggPos.Z - myPos.Z
			local distXZ = math.sqrt(dx * dx + dz * dz)

			if distXZ < 5 then break end

			local vel = root.AssemblyLinearVelocity
			local yVel = vel.Y
			if myPos.Y > eggPos.Y + 8 then
				yVel = math.min(yVel, 5)
			elseif myPos.Y < eggPos.Y - 2 then
				yVel = math.max(yVel, smallForce)
			else
				yVel = math.max(yVel, 12)
			end

			local dir = Vector3.new(dx, 0, dz).Unit
			root.AssemblyLinearVelocity = Vector3.new(dir.X * 85, yVel, dir.Z * 85)
			task.wait(0.04)
		end

		if not snipeActive then
			instantStop()
			return
		end

		local root = getRoot()
		if root and eggModel.Parent then
			eggPart = eggModel.PrimaryPart or eggModel:FindFirstChildWhichIsA("BasePart")
			if eggPart then
				local myPos = root.Position
				local eggPos = eggPart.Position
				local distXZ = (Vector3.new(myPos.X, 0, myPos.Z) - Vector3.new(eggPos.X, 0, eggPos.Z)).Magnitude
				local aboveEgg = (myPos.Y >= eggPos.Y - 3) and (distXZ < 10)

				instantStop()
				task.wait(0.05)
				if aboveEgg then
					tpToGround()
				end
			else
				instantStop()
			end
		else
			instantStop()
		end

		snipeActive = false
	end)
end

			lastRefresh = tick()
			pcall(updateAll)
		end
		task.wait(0.5)
	end
end)
